cmake_minimum_required(VERSION 3.16)
project(disp_fusion_fast LANGUAGES CXX)
option(DF_FAST_NATIVE_TUNING "Tune for the build host, not a portable DSP" OFF)
option(DF_FAST_PROFILE "Measure kernel stages with the host clock" OFF)
option(DF_FAST_SIMD "Allow compiler SIMD directives; does not create threads" ON)
option(DF_FAST_VECTOR_TYPES "Use optional GCC/Clang four-lane vector primitives" ON)
add_library(disp_fusion_fast SHARED fast_aligned.cpp)
target_compile_features(disp_fusion_fast PUBLIC cxx_std_17)
target_include_directories(disp_fusion_fast PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(DF_FAST_VECTOR_TYPES AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  include(CheckCXXSourceCompiles)
  check_cxx_source_compiles("typedef float F __attribute__((vector_size(16))); typedef int I __attribute__((vector_size(16))); int main(){I i{}; F f=__builtin_convertvector(i,F); return int(f[0]);}" DF_HAS_VECTOR_TYPES)
  if(DF_HAS_VECTOR_TYPES)
    target_compile_definitions(disp_fusion_fast PRIVATE DF_FAST_VECTOR_TYPES=1)
  endif()
endif()
if(DF_FAST_PROFILE)
  target_compile_definitions(disp_fusion_fast PRIVATE DF_FAST_PROFILE=1)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(disp_fusion_fast PRIVATE -O3 -fno-fast-math -fno-trapping-math -ffp-contract=off -Wall -Wextra -Wpedantic)
  if(DF_FAST_SIMD)
    include(CheckCXXCompilerFlag)
    check_cxx_compiler_flag("-fopenmp-simd" DF_HAS_SIMD_FLAG)
    if(DF_HAS_SIMD_FLAG)
      target_compile_options(disp_fusion_fast PRIVATE -fopenmp-simd)
      target_compile_definitions(disp_fusion_fast PRIVATE DF_FAST_SIMD=1)
    endif()
  endif()
  if(DF_FAST_NATIVE_TUNING)
    target_compile_options(disp_fusion_fast PRIVATE -march=native)
  endif()
endif()
add_executable(disp_fusion_fast_bench benchmark.cpp)
target_compile_features(disp_fusion_fast_bench PRIVATE cxx_std_17)
target_link_libraries(disp_fusion_fast_bench PRIVATE disp_fusion_fast)
install(TARGETS disp_fusion_fast LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES fast_aligned.h DESTINATION include)
